# 钢珠检测：水管ROI亮度列扫描（球沿管一维运动 → 一维检测）

def detect_ball(gray_np, img_np):
    """返回 (found, cx, cy, thr)；未找到 cx=-1"""
    roi = gray_np[PIPE_Y1:PIPE_Y2, PIPE_X1:PIPE_X2]
    mean = float(np.mean(roi))
    std  = float(np.std(roi))
    thr  = mean + THR_K * std
    if thr < GRAY_FLOOR:
        thr = GRAY_FLOOR

    mask = roi > thr
    col_counts = np.sum(mask, axis=0)
    COL_MIN = max(3, BALL_H_MIN // 2)

    best_cx = -1
    best_score = 0
    i = 0
    n = ROI_W
    while i < n:
        if int(col_counts[i]) >= COL_MIN:
            a = i
            while i < n and int(col_counts[i]) >= COL_MIN:
                i += 1
            b = i - 1
            w = b - a + 1
            if w < BALL_W_MIN or w > BALL_W_MAX:
                continue
            seg = mask[:, a:b+1]
            row_counts = np.sum(seg, axis=1)
            row_need = max(2, w // 3)
            h_extent = 0
            for r in range(ROI_H):
                if int(row_counts[r]) >= row_need:
                    h_extent += 1
            if h_extent < BALL_H_MIN:
                continue
            region = img_np[PIPE_Y1:PIPE_Y2, PIPE_X1+a:PIPE_X1+b+1]
            mr = int(np.mean(region[:, :, 0]))
            mb = int(np.mean(region[:, :, 2]))
            if (mr - mb) > COLOR_RB_DIFF:
                continue
            total = 0
            wsum = 0
            for x in range(a, b+1):
                c = int(col_counts[x])
                total += c
                wsum += c * x
            if total <= 0:
                continue
            if total > best_score:
                best_score = total
                best_cx = wsum / total + PIPE_X1
        else:
            i += 1

    if best_cx < 0:
        return False, -1, -1, thr
    return True, int(best_cx), (PIPE_Y1 + PIPE_Y2) // 2, thr

# ---------------- 跟踪滤波 ----------------
pos_hist = []
last_pos_mm = -1.0
lost_cnt = 0

def track_filter(found, cx):
    global last_pos_mm, lost_cnt
    if not found:
        lost_cnt += 1
        if last_pos_mm >= 0 and lost_cnt <= HOLD_FRAMES:
            return True, last_pos_mm
        return False, -1.0
    pos_mm = (cx - PIPE_X1) * PIPE_LEN_MM / ROI_W
    if pos_mm < 0: pos_mm = 0
    if pos_mm > PIPE_LEN_MM: pos_mm = PIPE_LEN_MM
    if last_pos_mm >= 0 and abs(pos_mm - last_pos_mm) > JUMP_MM:
        lost_cnt += 1
        if lost_cnt <= HOLD_FRAMES:
            return True, last_pos_mm
        return False, -1.0
    pos_hist.append(pos_mm)
    if len(pos_hist) > MEDIAN_N:
        pos_hist.pop(0)
    s = sorted(pos_hist)
    med = s[len(s) // 2]
    last_pos_mm = med
    lost_cnt = 0
    return True, med

# ---------------- UART（AA55校验帧） ----------------
# AA 55 PH PL FLAG CHK 0D   POS=mm*10   FLAG:1有效/0丢失   CHK=(PH+PL+FLAG)&0xFF
def uart_send(valid, pos_mm):
    try:
        val = int(pos_mm * 10) if valid else 0
        if val < 0: val = 0
        if val > 65535: val = 65535
        ph = (val >> 8) & 0xFF
        pl = val & 0xFF
        flag = 1 if valid else 0
        chk = (ph + pl + flag) & 0xFF
        uart.write(bytearray([0xAA, 0x55, ph, pl, flag, chk, 0x0D]))
    except Exception:
        pass

def _send_all(sock, data):
    view = memoryview(data)
    total = len(view)
    sent = 0
    retry = 0
    while sent < total:
        end = min(sent + SEND_CHUNK, total)
        try:
            n = sock.send(view[sent:end])
        except OSError as e:
            code = e.args[0] if e.args else 0
            if code in (11, 35):
                retry += 1
                if retry > 400:
                    raise
                time.sleep_ms(5)
                continue
            raise
        if not n:
            raise OSError("closed")
        sent += n
        retry = 0

class MjpegHub:
    def __init__(self):
        self._lock = _thread.allocate_lock()
        self._jpeg = None
        self._seq = 0
        self.connected = False      # 有客户端在拉流时为True（主循环据此决定是否编码）
    def publish(self, jpeg):
        self._lock.acquire()
        try:
            self._jpeg = jpeg
            self._seq += 1
        finally:
            self._lock.release()
    def fetch(self, last_seq):
        self._lock.acquire()
        try:
            if self._seq == last_seq or self._jpeg is None:
                return None, last_seq
            return self._jpeg, self._seq
        finally:
            self._lock.release()
    def latest(self):
        self._lock.acquire()
        try:
            return self._jpeg
        finally:
            self._lock.release()

def mjpeg_server():
    server = socket.socket()
    try:
        server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    except Exception:
        pass
    server.bind(("0.0.0.0", HTTP_PORT))
    server.listen(2)
    print("[HTTP] 端口:", HTTP_PORT)
    while True:
        client = None
        try:
            client, addr = server.accept()
            request = client.recv(512)
            if b"/stream.mjpg" in request:
                _send_all(client, b"HTTP/1.0 200 OK\r\nContent-Type: multipart/x-mixed-replace; boundary=frame\r\nCache-Control: no-store\r\nConnection: close\r\n\r\n")
                hub.connected = True
                last_seq = 0
                print("[HTTP] 客户端:", addr)
                while True:
                    jpeg, seq = hub.fetch(last_seq)
                    if jpeg is None:
                        time.sleep_ms(10)
                        os.exitpoint()
                        continue
                    last_seq = seq
                    _send_all(client, ("--frame\r\nContent-Type: image/jpeg\r\nContent-Length: {}\r\n\r\n".format(len(jpeg))).encode())
                    _send_all(client, jpeg)
                    _send_all(client, b"\r\n")
                    os.exitpoint()
            elif b"/snapshot" in request:
                jpeg = hub.latest()
                if jpeg is None:
                    _send_all(client, b"HTTP/1.0 503 Service Unavailable\r\nContent-Length: 0\r\nConnection: close\r\n\r\n")
                else:
                    _send_all(client, ("HTTP/1.0 200 OK\r\nContent-Type: image/jpeg\r\nContent-Length: {}\r\nConnection: close\r\n\r\n".format(len(jpeg))).encode())
                    _send_all(client, jpeg)
                client.close()
                client = None
            else:
                _send_all(client, ("HTTP/1.0 200 OK\r\nContent-Type: text/html; charset=utf-8\r\nContent-Length: {}\r\nConnection: close\r\n\r\n".format(len(HTML_PAGE))).encode())
                _send_all(client, HTML_PAGE)
                client.close()
                client = None
        except Exception:
            pass
        finally:
            hub.connected = False
            if client:
                try:
                    client.close()
                except Exception:
                    pass

def encode_jpeg(img):
    """编码JPEG并bytes拷贝（编码结果指向VB缓存，下一帧会覆盖）"""
    try:
        encoded = img.compressed(quality=JPEG_QUALITY)
    except Exception:
        encoded = img.compress(quality=JPEG_QUALITY)
    if encoded is None:
        return None
    return bytes(encoded.bytearray())
